function duplicateCount(text) {
    const lower = text.toLowerCase();
    const counts = {};
    
    for (let char of lower) {
        counts[char] = (counts[char] || 0) + 1;
    }
    
    return Object.values(counts).filter(count => count > 1).length;
}
